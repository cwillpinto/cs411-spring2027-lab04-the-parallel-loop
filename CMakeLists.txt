cmake_minimum_required(VERSION 3.18)
project(cs411_lab04_parallel_loop LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(transform src/transform.cpp)
target_include_directories(transform PUBLIC include)
target_link_libraries(transform PUBLIC OpenMP::OpenMP_CXX)
add_executable(lab4 src/main.cpp)
target_link_libraries(lab4 PRIVATE transform)
enable_testing()
add_executable(transform_tests tests/transform_tests.cpp)
target_link_libraries(transform_tests PRIVATE transform)
add_test(NAME transform_tests COMMAND transform_tests)
